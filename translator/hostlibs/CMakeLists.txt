cmake_minimum_required(VERSION 3.16)
if(NOT PROJECT_NAME)
  project(armtr_hostlibs C CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT TARGET armtr_proxy)
  set(ARMTR_PROXY_TESTS OFF)
  add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../proxy ${CMAKE_CURRENT_BINARY_DIR}/proxy_build)
endif()

add_library(armtr_hostlibs STATIC hostlibs.cpp)
target_include_directories(armtr_hostlibs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(armtr_hostlibs PUBLIC armtr_proxy ${CMAKE_DL_LIBS})
target_compile_options(armtr_hostlibs PRIVATE -Wall -Wextra)

if(NOT DEFINED ARMTR_HOSTLIBS_TESTS OR ARMTR_HOSTLIBS_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  if(NOT TARGET armtr_fake_engine)
    add_library(armtr_fake_engine STATIC ${CMAKE_CURRENT_SOURCE_DIR}/../proxy/tests/fake_engine.cpp)
    target_include_directories(armtr_fake_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../proxy/tests ${CMAKE_CURRENT_SOURCE_DIR}/../include)
  endif()
  # Fake HOST libraries (same file names as on Android) that record their calls.
  set(FAKEDIR ${CMAKE_CURRENT_BINARY_DIR}/fakelibs)
  foreach(l log z android jnigraphics egl gles2 sles mediandk vulkan)
    add_library(fake_${l} SHARED tests/fake_${l}.cpp)
    set_target_properties(fake_${l} PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${FAKEDIR})
  endforeach()
  set_target_properties(fake_log PROPERTIES OUTPUT_NAME log PREFIX lib)
  set_target_properties(fake_z PROPERTIES OUTPUT_NAME z PREFIX lib)
  set_target_properties(fake_android PROPERTIES OUTPUT_NAME android PREFIX lib)
  set_target_properties(fake_jnigraphics PROPERTIES OUTPUT_NAME jnigraphics PREFIX lib)
  set_target_properties(fake_egl PROPERTIES OUTPUT_NAME EGL PREFIX lib)
  set_target_properties(fake_gles2 PROPERTIES OUTPUT_NAME GLESv2 PREFIX lib)
  set_target_properties(fake_sles PROPERTIES OUTPUT_NAME OpenSLES PREFIX lib)
  set_target_properties(fake_vulkan PROPERTIES OUTPUT_NAME vulkan PREFIX lib)
  set_target_properties(fake_mediandk PROPERTIES OUTPUT_NAME mediandk PREFIX lib)
  foreach(l camera2ndk sync aaudio)   # same fake source as libmediandk, under the other sonames
    add_library(fake_${l} SHARED tests/fake_mediandk.cpp)
    set_target_properties(fake_${l} PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${FAKEDIR} OUTPUT_NAME ${l} PREFIX lib)
  endforeach()
  add_executable(test_hostlibs tests/test_hostlibs.cpp)
  target_compile_definitions(test_hostlibs PRIVATE FAKEDIR="${FAKEDIR}"
    IMPORTS_TXT="${CMAKE_CURRENT_SOURCE_DIR}/../loader/tests/yougram_imports.txt")
  target_link_libraries(test_hostlibs armtr_hostlibs armtr_fake_engine Threads::Threads)
  add_dependencies(test_hostlibs fake_log fake_z fake_android fake_jnigraphics fake_egl fake_gles2 fake_sles fake_mediandk
    fake_camera2ndk fake_sync fake_aaudio fake_vulkan)
  add_test(NAME hostlibs COMMAND test_hostlibs)
endif()
