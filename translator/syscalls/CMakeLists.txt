cmake_minimum_required(VERSION 3.13)
if(NOT CMAKE_PROJECT_NAME)
  project(armtr_syscalls C CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(armtr_syscalls STATIC syscalls.cpp)
target_include_directories(armtr_syscalls PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../include)
target_compile_options(armtr_syscalls PRIVATE -Wall -Wextra)
enable_testing()
foreach(t table file mem proc net)
  add_executable(armtr_sys_test_${t} tests/test_${t}.cpp)
  target_link_libraries(armtr_sys_test_${t} armtr_syscalls pthread)
  add_test(NAME syscalls_${t} COMMAND armtr_sys_test_${t})
endforeach()
